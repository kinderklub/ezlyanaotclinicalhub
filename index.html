Clinical Sub-App Manager (Workspace Modules)

Paste single-file HTML code or link an external URL for each OT sub-app. All sub-apps share one single Firebase connection via the postMessage bridge.

Sub-App Communication Bridge Protocol (window.postMessage)

Sub-apps inside an iframe do not need to import Firebase directly. Use the standard protocol below to save, load, or delete patient records through the Master Collector App: 

// 1. Save patient record to Master Collector App
window.parent.postMessage({
  type: 'OT_SAVE_RECORD',
  data: {
    docId: currentDocId || null, // null for new record, or existing docId to update
    module: 'neuro',             // Sub-app module ID (e.g., 'neuro', 'paeds', 'ortho')
    ptName: 'John Doe',
    ptMrn: 'RN-2026-8841',
    ptDate: '2026-11-20',
    ptDx: 'Left MCA Infarct (Right Hemiparesis)',
    summaryBadges: ['MBI: 65/100', 'RUE MMT: 3+/5', 'Goal: Independent Transfer'],
    payloadJson: { mbiScore: 65, clinicalNotes: 'Improved trunk control during dynamic sitting.' }
  }
}, '*');

// 2. Listen for save confirmation & record load commands (Open in Sub-App)
window.addEventListener('message', (event) => {
  const msg = event.data;
  if (msg?.type === 'OT_SAVE_SUCCESS') {
    currentDocId = msg.docId;
    console.log('Saved successfully with ID:', msg.docId);
  }
  if (msg?.type === 'OT_LOAD_RECORD') {
    const rec = msg.record;
    // Populate sub-app form fields using rec.ptName, rec.ptMrn, rec.payloadJson, etc.
  }
  if (msg?.type === 'OT_RECORDS_SYNC') {
    console.log('Synced records for this unit:', msg.records);
  }
});
